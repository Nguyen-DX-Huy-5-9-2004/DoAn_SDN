def _ensure_iproute2_web1(web1) -> None:
    """python:3.9-slim không có lệnh `ip` → Mininet/IpCompatIntf không bật được veth."""
    out = web1.cmd("bash -lc 'command -v ip && echo HAVE_IP'").strip()
    if out.endswith("HAVE_IP"):
        print("[DOCKER] web1: đã có lệnh `ip`.")
        return
    print("[DOCKER] web1: cài iproute2 (slim không có `ip`, lần đầu có thể ~30–90s)...")
    log = web1.cmd(
        "bash -lc 'export DEBIAN_FRONTEND=noninteractive && "
        "apt-get update -qq && apt-get install -y -qq iproute2 && command -v ip'"
    ).strip()
    tail = log[-200:] if len(log) > 200 else log
    print(f"[DOCKER] web1: apt-get kết thúc — {tail}")
def _ensure_iproute2_db1(db1) -> None:
    """Đủ lệnh `ip`/`ip address` tương thích Mininet (BusyBox ip thiếu tùy chọn)."""
    print("[DOCKER] db1: apk add iproute2...")
    log = db1.cmd(
        "sh -c 'apk add --no-cache iproute2 2>&1; command -v ip || true'"
    ).strip()
    tail = log[-200:] if len(log) > 200 else log
    print(f"[DOCKER] db1: {tail}")
